package main
 
import (
	"bufio"
	"os"
)
 
func main() {
	in := bufio.NewReaderSize(os.Stdin, 1<<20)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	readInt := func() int {
		n := 0
		c, _ := in.ReadByte()
		for c == ' ' || c == '
' || c == '\r' {
			c, _ = in.ReadByte()
		}
		for c >= '0' && c <= '9' {
			n = n*10 + int(c-'0')
			c, _ = in.ReadByte()
		}
		return n
	}
 
	t := readInt()
	for ; t > 0; t-- {
		n := readInt()
		odd := 0
		for i := 0; i < n; i++ {
			if readInt()%2 == 1 {
				odd++
			}
		}
		even := n - odd
 
		if (odd == 1 && even == 1) || (odd == 3 && even == 0) {
			out.WriteString("Larpity
")
		} else {
			out.WriteString("Moggity
")
		}
	}
}
