package main
 
import (
	"bufio"
	"fmt"
	"os"
	"strings"
)
 
func main() {
	reader := bufio.NewReader(os.Stdin)
	writer := bufio.NewWriter(os.Stdout)
	defer writer.Flush()
 
	var t int
	fmt.Fscan(reader, &t)
	for ; t > 0; t-- {
		var n int
		var s string
		fmt.Fscan(reader, &n, &s)
		if strings.Contains(s, "...") {
			fmt.Fprintln(writer, 2)
		} else {
			fmt.Fprintln(writer, strings.Count(s, "."))
		}
	}
}
