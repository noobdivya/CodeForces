package main
 
import (
	"bufio"
	"fmt"
	"os"
)
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
 
	for ; t > 0; t-- {
		var n int64
		fmt.Fscan(in, &n)
 
		var ans int64
 
		for b := int64(1); b <= n; b++ {
			x := n / b
			ans += x * x
		}
 
		fmt.Fprintln(out, ans)
	}
}
