package main
 
import (
	"bufio"
	"fmt"
	"os"
	"strings"
)
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
	for ; t > 0; t-- {
		var n, k int
		var s string
		fmt.Fscan(in, &n, &k, &s)
		ans := 0
		for i := 0; i < n; i += k {
			if !strings.Contains(s[i:i+k], "0") {
				ans++
			}
		}
		fmt.Fprintln(out, ans)
	}
}
