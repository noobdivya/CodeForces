package main
 
import (
	"bufio"
	"fmt"
	"os"
)
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
 
	for ; t > 0; t-- {
		var n, k, m int64
		fmt.Fscan(in, &n, &k, &m)
 
		if k > m {
			fmt.Fprintln(out, "NO")
			continue
		}
 
		fmt.Fprintln(out, "YES")
 
		for i := int64(1); i <= n; i++ {
			if i == k {
				fmt.Fprint(out, m-k+1)
			} else {
				fmt.Fprint(out, 1)
			}
 
			if i < n {
				fmt.Fprint(out, " ")
			}
		}
		fmt.Fprintln(out)
	}
}
