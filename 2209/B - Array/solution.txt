package main
 
import (
	"bufio"
	"fmt"
	"os"
)
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
	for ; t > 0; t-- {
		var n int
		fmt.Fscan(in, &n)
		a := make([]int, n)
		for i := range a {
			fmt.Fscan(in, &a[i])
		}
		for i := 0; i < n; i++ {
			less, greater := 0, 0
			for j := i + 1; j < n; j++ {
				if a[j] < a[i] {
					less++
				} else if a[j] > a[i] {
					greater++
				}
			}
			if greater > less {
				less = greater
			}
			fmt.Fprint(out, less, " ")
		}
		fmt.Fprintln(out)
	}
}
