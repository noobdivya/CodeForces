package main
 
import (
	"bufio"
	"fmt"
	"os"
)
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
 
	for ; t > 0; t-- {
		var n int
		var s string
 
		fmt.Fscan(in, &n)
		fmt.Fscan(in, &s)
 
		// Count original compressed length
		ans := 1
		for i := 1; i < n; i++ {
			if s[i] != s[i-1] {
				ans++
			}
		}
 
		// Find maximum reduction
		best := 0
 
		for i := 1; i < n-1; i++ {
			a := s[i-1]
			b := s[i]
			c := s[i+1]
 
			if a == c && a != b {
				best = 2
				break
			}
 
			if a != b && b != c && a != c {
				best = 1
			}
		}
 
		fmt.Fprintln(out, ans-best)
	}
}
