package main
 
import "fmt"
 
func main() {
	var t int
	fmt.Scan(&t)
	ans := []int{}
	// cout:=0
	for i := 0; i < t; i++ {
		var n int
		fmt.Scan(&n)
		arr := make([]int, n)
		for j := 0; j < n; j++ {
			fmt.Scan(&arr[j])
		}
		even := 0
		odd := 0
		for j := 0; j < n; j++ {
			if arr[j]%2 != j%2 {
				if arr[j]%2 == 0 {
					even++
				} else {
					odd++
				}
			}
		}
		if even != odd {
			ans = append(ans, -1)
		} else {
			ans = append(ans, even)
		}
	}
	for i := 0; i < t; i++ {
		fmt.Println(ans[i])
	}
}
