package main
 
import (
	"bufio"
	"fmt"
	"os"
)
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
	for ; t > 0; t-- {
		var n, k int
		fmt.Fscan(in, &n, &k)
		if k < n || k > 2*n-1 {
			fmt.Fprintln(out, -1)
			continue
		}
		d := 2*n - k // number of cells that are both row min and column min
 
		a := make([][]int, n)
		for i := range a {
			a[i] = make([]int, n)
		}
 
		cnt := 1
		for i := 0; i < d; i++ { // doubles on the diagonal
			a[i][i] = cnt
			cnt++
		}
		for i := d; i < n; i++ { // row mins only
			a[i][0] = cnt
			cnt++
		}
		for j := d; j < n; j++ { // column mins only
			a[0][j] = cnt
			cnt++
		}
		for i := 0; i < n; i++ { // everything else is larger
			for j := 0; j < n; j++ {
				if a[i][j] == 0 {
					a[i][j] = cnt
					cnt++
				}
			}
		}
 
		for i := 0; i < n; i++ {
			for j := 0; j < n; j++ {
				fmt.Fprint(out, a[i][j], " ")
			}
			fmt.Fprintln(out)
		}
	}
}
