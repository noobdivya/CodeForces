package main
 
import "fmt"
 
func main() {
	var n int
	fmt.Scan(&n)
	a := make([]int, n)
	b := make([]int, n)
	for i := 0; i < n; i++ {
		fmt.Scan(&a[i], &b[i])
	}
	ans := 0
	for i := 0; i < n; i++ {
		for j := 0; j < n; j++ {
			if i != j && a[i] == b[j] {
				ans++
			}
		}
	}
	fmt.Println(ans)
}
