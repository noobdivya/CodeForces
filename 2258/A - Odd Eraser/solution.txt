package main
 
import (
	"bufio"
	"fmt"
	"os"
)
 
func gcd(a, b int) int {
	for b != 0 {
		a, b = b, a%b
	}
	return a
}
 
func main() {
	in := bufio.NewReader(os.Stdin)
	out := bufio.NewWriter(os.Stdout)
	defer out.Flush()
 
	var t int
	fmt.Fscan(in, &t)
	for ; t > 0; t-- {
		var n int
		fmt.Fscan(in, &n)
		a := make([]int, n)
		for i := range a {
			fmt.Fscan(in, &a[i])
		}
		fmt.Fprintln(out, gcd(a[0], a[n-1]))
	}
}
